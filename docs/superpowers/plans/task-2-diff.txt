 impl FieldMap for mlua::Table {
     fn get_field(&self, key: &str) -> anyhow::Result<Cow<str>> {
         self.get::<String>(key)
             .map(|s| Cow::Owned(s))
             .map_err(|e| anyhow!("{}", e))
     }
 }
