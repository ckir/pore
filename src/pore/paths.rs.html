1//! Base-directory resolution for the config file and the index cache.
2//!
3//! pore follows the XDG layout, but `HOME` — which the XDG fallback depends on — is not
4//! a standard Windows environment variable. Windows sets `USERPROFILE`; `HOME` exists
5//! only if something else put it there, such as Git Bash or WSL. Depending on `HOME`
6//! alone made every command that touches config or cache fail on a stock Windows box
7//! with a bare "environment variable not found".
8
9use anyhow::anyhow;
10use std::env;
11use std::ffi::OsString;
12use std::path::PathBuf;
13
14/// Returns the value of `var` if it is set and non-empty.
15///
16/// An empty variable is treated as unset: an exported-but-blank `XDG_CONFIG_HOME` would
17/// otherwise resolve the config path to the process's current directory.
18fn non_empty_var(var: &str) -> Option<OsString> {
19    env::var_os(var).filter(|v| !v.is_empty())
20}
21
22/// Resolves a base directory, preferring `xdg_var` and falling back to `<home>/<suffix>`.
23///
24/// `HOME` is consulted before `USERPROFILE` so that an explicitly set `HOME` keeps
25/// winning. That is what every existing install relies on, including Git Bash and WSL,
26/// where the two point at different places.
27///
28/// # Errors
29///
30/// Returns an error naming every variable it consulted when none of them is set, rather
31/// than propagating `VarError::NotPresent`, which renders as "environment variable not
32/// found" and names neither the variable nor a remedy.
33pub fn resolve_base_dir(xdg_var: &str, suffix: &str) -> Result<PathBuf, anyhow::Error> {
34    if let Some(dir) = non_empty_var(xdg_var) {
35        return Ok(PathBuf::from(dir));
36    }
37    for var in ["HOME", "USERPROFILE"] {
38        if let Some(home) = non_empty_var(var) {
39            return Ok(PathBuf::from(home).join(suffix));
40        }
41    }
42    Err(anyhow!(
43        "cannot determine the {suffix} directory: none of {xdg_var}, HOME or USERPROFILE is set. \
44         Set {xdg_var} to choose the location explicitly."
45    ))
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51
52    /// These mutate process-wide environment state, so they are kept in one test to
53    /// avoid racing the other tests in this binary.
54    #[test]
55    fn resolution_order_and_error() {
56        let saved: Vec<_> = ["PORE_TEST_XDG", "HOME", "USERPROFILE"]
57            .iter()
58            .map(|v| (*v, env::var_os(v)))
59            .collect();
60
61        env::remove_var("PORE_TEST_XDG");
62        env::remove_var("HOME");
63        env::remove_var("USERPROFILE");
64
65        // Nothing set: the error names each variable consulted.
66        let err = resolve_base_dir("PORE_TEST_XDG", ".config")
67            .expect_err("no variables set must not resolve")
68            .to_string();
69        assert!(err.contains("PORE_TEST_XDG"), "{err}");
70        assert!(err.contains("USERPROFILE"), "{err}");
71
72        // USERPROFILE alone is enough -- this is the Windows case that used to fail.
73        env::set_var("USERPROFILE", "/up");
74        assert_eq!(
75            resolve_base_dir("PORE_TEST_XDG", ".config").unwrap(),
76            PathBuf::from("/up").join(".config")
77        );
78
79        // HOME wins over USERPROFILE.
80        env::set_var("HOME", "/home");
81        assert_eq!(
82            resolve_base_dir("PORE_TEST_XDG", ".config").unwrap(),
83            PathBuf::from("/home").join(".config")
84        );
85
86        // The XDG variable wins over both, and is used as-is with no suffix.
87        env::set_var("PORE_TEST_XDG", "/xdg");
88        assert_eq!(
89            resolve_base_dir("PORE_TEST_XDG", ".config").unwrap(),
90            PathBuf::from("/xdg")
91        );
92
93        // An empty value counts as unset, or the path would resolve to the cwd.
94        env::set_var("PORE_TEST_XDG", "");
95        assert_eq!(
96            resolve_base_dir("PORE_TEST_XDG", ".config").unwrap(),
97            PathBuf::from("/home").join(".config")
98        );
99
100        for (var, val) in saved {
101            match val {
102                Some(v) => env::set_var(var, v),
103                None => env::remove_var(var),
104            }
105        }
106    }
107}
