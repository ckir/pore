1//! Configuration file loading and option merging.
2//!
3//! This module handles loading the TOML config file from `$XDG_CONFIG_HOME/pore.toml`
4//! (falling back to `$HOME/.config/pore.toml`) and merging options from three sources:
5//!
6//! 1. **Global defaults** — hardcoded `Default` implementations.
7//! 2. **Config file globals** — top-level TOML keys apply to all directories.
8//! 3. **Local config sections** — TOML tables keyed by an arbitrary name that contain a
9//!    `path` field matching the query directory. Local sections may optionally nest an
10//!    index-specific subsection (looked up by `index_name`).
11//!
12//! # Option types
13//!
14//! `SearchConfigOpt` and `FileIndexOptionsShape` are "optional" shapes where every field
15//! is `Option<T>`. They are produced by the `#[create_option_copy]` macro and allow
16//! incremental merging: only explicitly-set CLI flags or config-file keys populate a value,
17//! so defaults from the config file do not accidentally override more specific settings.
18

32/// Resolved search configuration with concrete (non-optional) defaults.
33///
34/// All fields have sensible defaults and are populated either from the config file or
35/// from `SearchConfig::default()`. The `#[create_option_copy]` macro generates the
36/// companion `SearchConfigOpt` struct (with `Option<T>` fields) and a `merge_from` method.
37#[create_option_copy(SearchConfigOpt)]
38#[derive(Debug, Deserialize, Clone)]
39pub struct SearchConfig {
40    /// Output results as JSON instead of human-readable text.
41    pub json: bool,
42    /// Maximum number of files to return.
43    pub limit: usize,
44    /// Minimum relevance score threshold for including a result.
45    pub threshold: f32,
46    /// When true, only print matching filenames, not the matching lines.
47    pub filename_only: bool,
48    /// Color output mode (auto, always, ansi, never).
49    pub color: ColorMode,
50    /// Force a full index rebuild before searching.
51    pub rebuild_index: bool,
52    /// Whether to update the index before searching (default: true).
53    pub update: bool,
54    /// When true, keep the index in memory only (no disk cache).
55    pub in_memory: bool,
56}
57
58impl Default for SearchConfig {
59    fn default() -> SearchConfig {
60        SearchConfig {
61            json: false,
62            limit: 1000,
63            threshold: 0.0,
64            filename_only: false,
65            color: ColorMode::Auto,
66            rebuild_index: false,
67            update: true,
68            in_memory: false,
69        }
70    }
71}

73impl SearchConfig {
74    /// Convert search settings into a `FileSearchOptions` for executing a query.
75    ///
76    /// `search_dir` is the subdirectory within the index to restrict results to.
77    pub fn to_opts(&self, search_dir: &str) -> FileSearchOptions {
78        FileSearchOptions {
79            limit: self.limit,
80            threshold: self.threshold,
81            filename_only: self.filename_only,
82            root_dir: Some(search_dir.to_string()),
83        }
84    }
85}

87/// Load configuration from the TOML config file and merge with defaults.
88///
89/// The config file is looked up at `$XDG_CONFIG_HOME/pore.toml` or `$HOME/.config/pore.toml`.
90/// If the file does not exist, default options are returned.
91///
92/// # Merging order
93///
94/// 1. Start from global top-level TOML keys.
95/// 2. Find a local config section whose `path` matches `path`. Merge it.
96/// 3. If `index_name` is provided, look for a matching subsection within the local config
97///    (e.g. `[local-1.my_index]`). Merge it.
98/// 4. If `index_name` is provided but no local section matched, look for a global named
99///    index (`[index-<name>]`). Merge it.
100///
101/// # Errors
102///
103/// Returns an error if the config file exists but cannot be parsed, or if `index_name`
104/// is specified but no matching index configuration is found.
105pub fn load_config(
106    path: &Path,
107    index_name: Option<&str>,
108) -> Result<(FileIndexOptionsShape, SearchConfigOpt), anyhow::Error> {
109    let path_str = path.to_string_lossy();
110    let mut config_home = env::var("XDG_CONFIG_HOME").unwrap_or("".to_string());
111    if config_home.is_empty() {
112        config_home = env::var("HOME")? + "/.config";
113    }
114    let config_file = PathBuf::from(config_home).join(CONFIG_FILE);
115    if config_file.exists() {
116        let contents = &fs::read_to_string(&config_file)?;
117        let value: Value = toml::from_str(contents)
118            .map_err(|e| anyhow!("Error parsing config file {:?}: {}", config_file, e))?;
119        let mut index: FileIndexOptionsShape = value.clone().try_into()?;
120        let mut search: SearchConfigOpt = value.clone().try_into()?;
121
122        let mut found_index = false;
123        if let Value::Table(table) = &value {
124            // Look for a local configuration with a matching path
125            for (_, val) in table.iter() {
126                if let Value::Table(local_config) = val {
127                    if local_config.get("path") == Some(&Value::String(path_str.to_string())) {
128                        index.merge_from(&val.clone().try_into()?);
129                        search.merge_from(&val.clone().try_into()?);
130                        // Look for an index the local config
131                        if let Some(idx_name) = index_name {
132                            if let Some(local_index_config) = local_config.get(idx_name) {
133                                index.merge_from(&local_index_config.clone().try_into()?);
134                                search.merge_from(&local_index_config.clone().try_into()?);
135                                found_index = true;
136                            }
137                        }
138                        break;
139                    }
140                }
141            }
142            // if index exists, find global index and load it
143            if let Some(name) = index_name {
144                if let Some(global_index) = table.get(&format!("index-{}", name)) {
145                    index.merge_from(&global_index.clone().try_into()?);
146                    search.merge_from(&global_index.clone().try_into()?);
147                    found_index = true;
148                }
149            }
150        }
151        if let Some(name) = index_name {
152            if !found_index {
153                bail!("Could not find index '{}'", name);
154            }
155        }
156
157        return Ok((index, search));
158    }
159    Ok((FileIndexOptionsShape::default(), SearchConfigOpt::default()))
160}

162#[cfg(test)]
163mod tests {
164    use std::{env, fs, path::PathBuf, str::FromStr};

166    use pore_core::FileIndexOptions;

172    #[test]
173    fn parsing_opt_configs_works() {
174        let contents = "follow = false
175threads = 100
176limit = 4
177";
178        let index: FileIndexOptionsShape = toml::from_str(contents).unwrap();
179        assert_eq!(index.follow, Some(false));
180        assert_eq!(index.threads, Some(100));
181        assert_eq!(index.ignore_files, None);
182        let search: SearchConfigOpt = toml::from_str(contents).unwrap();
183        assert_eq!(search.limit, Some(4));
184        assert_eq!(search.json, None);
185    }
186
187    #[test]
188    fn merging_opt_configs_works() {
189        let mut i1 = FileIndexOptionsShape {
190            follow: Some(true),
191            ..Default::default()
192        };
193        let i2 = FileIndexOptionsShape {
194            threads: Some(20),
195            ..Default::default()
196        };
197        i1.merge(&i2);
198        assert_eq!(i1.follow, Some(true));
199        assert_eq!(i1.threads, Some(20));
200        assert_eq!(i1.language, None);
201        let conf: FileIndexOptions = i1.into();
202        assert!(conf.follow);
203        assert_eq!(conf.threads, 20);
204        assert!(!conf.hidden);
205    }

207    #[test]
208    fn can_load_and_merge_defaults() {
209        let tmpdir = tempfile::tempdir().unwrap();
210        env::set_var("XDG_CONFIG_HOME", tmpdir.path().as_os_str());
211        let conf_file = PathBuf::from(tmpdir.path()).join(CONFIG_FILE);
212        fs::write(
213            conf_file,
214            r#"threads = 10
215
216[index-global_index]
217threads = 20
218
219[local-1]
220path = '/foo'
221threads = 30
222
223[local-1.local_index]
224threads = 40
225"#,
226        )
227        .unwrap();
228
229        let (index, _) = load_config(&PathBuf::from_str("/").unwrap(), None).unwrap();
230        assert_eq!(index.threads, Some(10));
231        let (index, _) =
232            load_config(&PathBuf::from_str("/").unwrap(), Some("global_index")).unwrap();
233        assert_eq!(index.threads, Some(20));
234        let (index, _) = load_config(&PathBuf::from_str("/foo").unwrap(), None).unwrap();
235        assert_eq!(index.threads, Some(30));
236        let (index, _) =
237            load_config(&PathBuf::from_str("/foo").unwrap(), Some("local_index")).unwrap();
238        assert_eq!(index.threads, Some(40));
239    }

241    #[test]
242    fn example_file_is_complete() {
243        let example = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("pore.example.toml");
244        let contents = &fs::read_to_string(&example).unwrap();
245        let value: toml::Value = toml::from_str(contents)
246            .unwrap_or_else(|_| panic!("Error parsing config file {:?}", example));
247        let index: FileIndexOptionsShape = value.clone().try_into().unwrap();
248        let search: SearchConfigOpt = value.clone().try_into().unwrap();
249        if let Err(missing_fields) = index.all() {
250            panic!("pore.example.toml is missing fields: {:?}", missing_fields);
251        }
252        if let Err(missing_fields) = search.all() {
253            panic!("pore.example.toml is missing fields: {:?}", missing_fields);
254        }
255    }
