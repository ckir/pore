1//! CLI argument parsing for the `pore` binary.
2//!
3//! This module defines the [`GlobalConfig`] struct that holds the fully-resolved
4//! configuration after merging CLI flags, and the [`parse_args`] function that builds
5//! it using `clap`.
6//!
7//! # Argument groups
8//!
9//! Arguments are split into two conceptual groups:
10//! - **Index options** — control which files are indexed and how (hidden files, symlinks,
11//!   languages, globs, threading). These conflict with `--index` because a named index
12//!   carries its own settings from the config file.
13//! - **Search options** — control query behavior (limit, threshold, output format, colors).
14//!
15//! Four mutually exclusive commands are available via flag groups:
16//! `--files`, `--indexes`, `--delete`, or a plain search (default).
17
18use std::path::{Path, PathBuf};
19use std::str::FromStr;
20use std::{env, fs};
21
22use clap::ArgGroup;
23use clap::{Arg, Command};
24use pore_core::language::LanguageRef;
25use pore_core::FileIndexOptionsShape;
26
27use crate::color_mode::ColorMode;
28use crate::config::SearchConfigOpt;
29
30/// The command to execute after configuration is resolved.
31#[derive(Debug)]
32pub enum CmdArg {
33    /// Execute a search query.
34    Search,
35    /// Print the files that would be searched, then exit.
36    ListFiles,
37    /// Print index metadata, then exit.
38    ListIndex,
39    /// Delete cached index files for the directory.
40    Delete,
41}
42
43/// Fully-resolved configuration after parsing CLI arguments and loading the config file.
44///
45/// `index` and `search` contain the options parsed from flags (as `*Shape` option structs
46/// with `Option<T>` fields), which are later merged with defaults and config-file values
47/// before conversion into the final `FileIndexOptions` and [`SearchConfig`](crate::config::SearchConfig).
48#[derive(Debug)]
49pub struct GlobalConfig {
50    /// Index-related options (before merging with config-file defaults).
51    pub index: FileIndexOptionsShape,
52    /// Search-related options (before merging with config-file defaults).
53    pub search: SearchConfigOpt,
54    /// The command to execute.
55    pub command: CmdArg,
56    /// The search query string, if provided.
57    pub query: Option<String>,
58    /// The directory to index/search within.
59    pub query_path: PathBuf,
60    /// A subdirectory within the index to restrict the search to.
61    pub search_dir: String,
62    /// Named index to use (from `--index`), if any.
63    pub index_name: Option<String>,
64}
65
66/// Parse command-line arguments and return a [`GlobalConfig`].
67///
68/// This function builds the full clap `Command`, extracts index and search options from
69/// the matches, determines which command to run, and resolves the query path.
70///
71/// # Errors
72///
73/// Returns an error if argument parsing fails (e.g., invalid values for `--threads`,
74/// `--limit`, or `--threshold`), or if path canonicalization fails.
75pub fn parse_args() -> Result<GlobalConfig, anyhow::Error> {
76    let matches = Command::new(env!("CARGO_PKG_NAME"))
77        .version(env!("CARGO_PKG_VERSION"))
78        .about(env!("CARGO_PKG_DESCRIPTION"))
79        .author(env!("CARGO_PKG_AUTHORS"))
80        // Index args
81        .arg(
82            Arg::new("index")
83                .short('i')
84                .long("index")
85                .num_args(1)
86                .conflicts_with_all(["in_memory", "no_memory", "hidden", "no_hidden", "follow_links", "no_follow_links", "language", "glob", "oglob", "glob_case_insensitive"])
87                .help("Use the specified index for querying (must be specified in the config file)")
88        )
89        .arg(
90            Arg::new("update")
91                .short('u')
92                .long("update")
93                .action(clap::ArgAction::SetTrue)
94                .help("Update the index before searching (the default)"),
95        )
96        .arg(
97            Arg::new("no_update")
98                .long("no-update")
99                .action(clap::ArgAction::SetTrue)
100                .conflicts_with("update")
101                .help("Do not update the index before performing the query"),
102        )
103        .arg(
104            Arg::new("in_memory")
105                .long("in-memory")
106                .action(clap::ArgAction::SetTrue)
107                .help("Do not store the text index on disk (will have to rebuild every time)"),
108        )
109        .arg(
110            Arg::new("no_memory")
111                .long("no-memory")
112                .action(clap::ArgAction::SetTrue)
113                .conflicts_with("in_memory")
114                .help("Force the index to be saved to disk (overrides --in-memory)"),
115        )
116        .arg(
117            Arg::new("hidden")
118                .long("hidden")
119                .action(clap::ArgAction::SetTrue)
120                .help("Search hidden files and directories"),
121        )
122        .arg(
123            Arg::new("no_hidden")
124                .long("no-hidden")
125                .action(clap::ArgAction::SetTrue)
126                .conflicts_with("hidden")
127                .help("Ignore hidden files and directories (overrides --hidden)"),
128        )
129        .arg(
130            Arg::new("follow_links")
131                .short('L')
132                .long("follow")
133                .action(clap::ArgAction::SetTrue)
134                .help("Follow symbolic links"),
135        )
136        .arg(
137            Arg::new("no_follow_links")
138                .long("no-follow")
139                .action(clap::ArgAction::SetTrue)
140                .conflicts_with("follow_links")
141                .help("Don't follow symbolic links (overrides --follow)"),
142        )
143        .arg(
144            Arg::new("language")
145                .long("language")
146                .value_parser(LanguageRef::from_str)
147                .help("The language to use for parsing files"),
148        )
149        .arg(
150            Arg::new("glob")
151                .short('g')
152                .long("glob")
153                .help("Include or exclude files and directories for searching that match the given glob. This always overrides any other ignore logic. Multiple glob flags may be used. Precede a glob with a ! to exclude it.")
154                .value_delimiter(',')
155                .num_args(1..)
156        )
157        .arg(
158            Arg::new("oglob")
159                .long("oglob")
160                .help("Only search files that match this glob. Files that do not match any of these globs will be ignored.")
161                .value_delimiter(',')
162                .num_args(1..)
163        )
164        .arg(
165            Arg::new("glob_case_insensitive")
166                .long("glob-case-insensitive")
167                .action(clap::ArgAction::SetTrue)
168                .help("Patterns passed to --glob and --oglob will be matched in a case-insentive way.")
169        )
170        // Index args that don't conflict with --index
171        .arg(
172            Arg::new("threads")
173                .short('j')
174                .long("threads")
175                .num_args(1)
176                .value_parser(|a: &str| a.parse::<usize>().map_err(|_|"threads must be an unsigned integer".to_string()))
177                .help("The approximate number of threads to use. A value of 0 (which is the default) will choose the thread count using heuristics.")
178        )
179        .arg(
180            Arg::new("rebuild_index")
181            .long("rebuild")
182            .action(clap::ArgAction::SetTrue)
183            .help("Force rebuild the index before searching")
184        )
185
186        // Search args
187        .arg(
188            Arg::new("limit")
189                .long("limit")
190                .num_args(1)
191                .value_parser(|a: &str| a.parse::<usize>().map_err(|_|"limit must be an unsigned integer".to_string()))
192                .help("Maximum number of files to return"),
193        )
194        .arg(
195            Arg::new("threshold")
196                .long("threshold")
197                .num_args(1)
198                .value_parser(|a: &str| a.parse::<f32>().map_err(|_|"threshold must be a floating point number".to_string()))
199                .help("Minimum score threshold for results"),
200        )
201        .arg(
202            Arg::new("json")
203                .long("json")
204                .action(clap::ArgAction::SetTrue)
205                .conflicts_with("commands")
206                .help("Print the results as json"),
207        )
208        .arg(
209            Arg::new("files_with_matches")
210                .short('l')
211                .long("files-with-matches")
212                .action(clap::ArgAction::SetTrue)
213                .conflicts_with("commands")
214                .help("Print out the files that match the search (not the matching lines)."),
215        )
216        .arg(
217            Arg::new("no_ignore")
218                .long("no-ignore")
219                .action(clap::ArgAction::SetTrue)
220                .help("Don't respect .gitignore files"),
221        )
222        .arg(
223            Arg::new("color")
224                .long("color")
225                .num_args(1)
226                .value_parser(["never", "auto", "always", "ansi"])
227                .hide_possible_values(true)
228                .help("This flag controls when to use colors. The default setting is auto, which will try to guess when to use colors.")
229                .long_help("This flag controls when to use colors. The default setting is auto, which will try to guess when to use colors.
230   The possible values for this flag are:
231
232       never    Colors will never be used.
233       auto     Auto-detect if the terminal supports colors (default).
234       always   Colors will always be used regardless of where output is sent.
235       ansi     Like 'always', but emits ANSI escapes (even in a Windows console).")
236        )
237        .group(
238            ArgGroup::new("commands")
239             .args(["files", "indexes", "delete"])
240            )
241        .arg(
242            Arg::new("files")
243                .long("files")
244                .action(clap::ArgAction::SetTrue)
245                .help("Print out the files that would be searched (do not perform the search)"),
246        )
247        .arg(
248            Arg::new("indexes")
249                .long("indexes")
250                .action(clap::ArgAction::SetTrue)
251                .help("print out the indexes that would be used (do not perform the search)")
252        )
253        .arg(
254            Arg::new("delete")
255                .long("delete")
256                .action(clap::ArgAction::SetTrue)
257                .help("Delete the cached index files for the directory (if any)")
258        )
259        .arg(Arg::new("query"))
260        .arg(Arg::new("dir"))
261        .get_matches();
262
263    let mut index = FileIndexOptionsShape::default();
264    // Parse index options
265    if matches.contains_id("hidden") && matches.get_flag("hidden") {
266        index.hidden = Some(true);
267    } else if matches.contains_id("no_hidden") && matches.get_flag("no_hidden") {
268        index.hidden = Some(false);
269    }
270    if let Some(lang) = matches.get_one::<LanguageRef>("language") {
271        index.language = Some(*lang);
272    }
273    if matches.contains_id("follow_links") && matches.get_flag("follow_links") {
274        index.follow = Some(true);
275    } else if matches.contains_id("no_follow_links") && matches.get_flag("no_follow_links") {
276        index.follow = Some(false);
277    }
278    if matches.contains_id("no_ignore") && matches.get_flag("no_ignore") {
279        index.ignore_files = Some(false);
280    }
281    if matches.contains_id("glob_case_insensitive") && matches.get_flag("glob_case_insensitive") {
282        index.glob_case_insensitive = Some(true);
283    }
284    if let Some(globs) = matches.get_many::<String>("glob") {
285        index.glob = Some(globs.map(|s| s.to_string()).collect());
286    }
287    if let Some(globs) = matches.get_many::<String>("oglob") {
288        index.oglob = Some(globs.map(|s| s.to_string()).collect());
289    }
290    if let Some(threads) = matches.get_one::<usize>("threads") {
291        index.threads = Some(*threads);
292    }
293
294    // Parse search options
295    let mut search = SearchConfigOpt::default();
296    if matches.contains_id("json") && matches.get_flag("json") {
297        search.json = Some(true);
298    }
299    if let Some(limit) = matches.get_one::<usize>("limit") {
300        search.limit = Some(*limit);
301    }
302    if let Some(threshold) = matches.get_one::<f32>("threshold") {
303        search.threshold = Some(*threshold);
304    }
305    if matches.contains_id("files_with_matches") && matches.get_flag("files_with_matches") {
306        search.filename_only = Some(true);
307    }
308    if let Some(color) = matches.get_one::<String>("color") {
309        search.color = Some(ColorMode::from_str(color).unwrap());
310    }
311    if matches.contains_id("rebuild_index") && matches.get_flag("rebuild_index") {
312        search.rebuild_index = Some(true);
313    }
314    if matches.contains_id("no_update") && matches.get_flag("no_update") {
315        search.update = Some(false);
316    } else if matches.contains_id("update") && matches.get_flag("update") {
317        search.update = Some(true);
318    };
319    if matches.contains_id("in_memory") && matches.get_flag("in_memory") {
320        search.in_memory = Some(true);
321    } else if matches.contains_id("no_memory") && matches.get_flag("no_memory") {
322        search.in_memory = Some(false);
323    }
324
325    let mut command = CmdArg::Search;
326    if matches.contains_id("delete") && matches.get_flag("delete") {
327        command = CmdArg::Delete;
328    } else if matches.contains_id("files") && matches.get_flag("files") {
329        command = CmdArg::ListFiles;
330    } else if matches.contains_id("indexes") && matches.get_flag("indexes") {
331        command = CmdArg::ListIndex;
332    }
333    let search_dir = matches
334        .get_one::<String>("dir")
335        .cloned()
336        .unwrap_or_default();
337    let query_path = if search_dir.is_empty() {
338        env::current_dir()?
339    } else {
340        fs::canonicalize(Path::new(&search_dir))?
341    };
342
343    Ok(GlobalConfig {
344        index,
345        search,
346        command,
347        query: matches.get_one::<String>("query").cloned(),
348        query_path,
349        search_dir,
350        index_name: matches.get_one::<String>("index").cloned(),
351    })
352}
