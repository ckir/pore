1//! File-based full-text indexing.
2//!
3//! This module provides [`FileIndex`], which indexes the text contents of files
4//! in a directory tree. It uses the `ignore` crate for efficient parallel
5//! directory walking that respects `.gitignore`, `.gitexclude`, and other
6//! ignore files.
7//!
8//! # Key types
9//! - [`FileIndex`] — the index handle, created via [`FileIndex::get_or_create`].
10//! - [`FileIndexOptions`] — configuration for file walking and tokenization.
11//! - [`FileSearchOptions`] — parameters controlling search result limits and format.
12//! - [`FileMetadata`] — persisted metadata about when and how the index was built.
13//! - [`FileSearchResult`] and [`Line`] — search result structures.
14

45/// A file-based full-text index.
46///
47/// Holds a Tantivy [`Index`] along with the schema fields (`filepath`, `contents`)
48/// and metadata about the indexed directory. Created via [`FileIndex::get_or_create`].
49#[derive(Debug, Clone)]
50pub struct FileIndex {
51    meta: FileMetadata,
52    cache_dir: Option<PathBuf>,
53    index: Index,
54    filepath: Field,
55    contents: Field,
56}

58/// Configuration options for building a [`FileIndex`].
59///
60/// Controls which files are included, whether to follow symlinks, the stemming
61/// language, and how many threads to use for the directory walker.
62///
63/// The `#[create_option_copy]` macro generates a companion `*Shape` struct and
64/// copy-conversion functions for Lua interop.
65#[create_option_copy(FileIndexOptionsShape)]
66#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
67pub struct FileIndexOptions {
68    /// Whether to follow symbolic links during file walking.
69    pub follow: bool,
70    /// Glob patterns to include files.
71    pub glob: Vec<String>,
72    /// Whether glob matching is case-insensitive.
73    pub glob_case_insensitive: bool,
74    /// Whether to include hidden files and directories.
75    pub hidden: bool,
76    /// Whether to respect .gitignore and similar ignore files.
77    pub ignore_files: bool,
78    /// Language used for stemming.
79    pub language: LanguageRef,
80    /// Glob patterns to exclude files (takes precedence over `glob`).
81    pub oglob: Vec<String>,
82    // TODO: move this field to a more appropriate location.
83    /// Number of threads for parallel directory walking (0 = auto).
84    pub threads: usize,
85}
86
87impl Default for FileIndexOptions {
88    fn default() -> FileIndexOptions {
89        FileIndexOptions {
90            follow: false,
91            hidden: false,
92            language: LanguageRef::English,
93            ignore_files: true,
94            glob_case_insensitive: false,
95            glob: vec![],
96            oglob: vec![],
97            threads: 0,
98        }
99    }
100}

105pub struct FileSearchOptions {

145impl FileMetadata {
146    /// Creates new metadata for the given directory.
147    ///
148    /// The path is canonicalized to an absolute path.
149    pub fn new<P: AsRef<Path>>(
150        config: FileIndexOptions,
151        for_dir: P,
152    ) -> Result<Self, anyhow::Error> {
153        let path = for_dir.as_ref();
154        Ok(FileMetadata {
155            config,
156            version: env!("CARGO_PKG_VERSION").to_string(),
157            last_update: DateTime::<Utc>::from_naive_utc_and_offset(
158                chrono::DateTime::<Utc>::UNIX_EPOCH.naive_utc(),
159                Utc,
160            ),
161            for_dir: fs::canonicalize(if path.is_absolute() {
162                path.to_path_buf()
163            } else {
164                env::current_dir()?.join(path)
165            })?,
166        })
167    }

254impl FileIndex {
255    /// Returns a reference to the underlying Tantivy index.
256    pub fn index(&self) -> &Index {
257        &self.index
258    }
259    /// Returns the schema field for file paths.
260    pub fn filepath(&self) -> &Field {
261        &self.filepath
262    }
263    /// Returns the schema field for file contents.
264    pub fn contents(&self) -> &Field {
265        &self.contents
266    }
267    /// Deletes the index and its on-disk cache (if any).
268    pub fn delete(&self) -> anyhow::Result<bool> {
269        delete_index(&self.index, self.cache_dir.as_deref())
270    }
271    /// Opens an existing index or creates a new one.
272    ///
273    /// If a persisted index exists at `cache_dir` with a matching config, it is
274    /// reused. Otherwise a fresh index is created.
275    ///
276    /// # Parameters
277    /// * `for_dir` — the directory to index.
278    /// * `cache_dir` — optional path for persisted index files.
279    /// * `config` — indexing options.
280    pub fn get_or_create<P: AsRef<Path>>(
281        for_dir: P,
282        cache_dir: Option<P>,
283        config: &FileIndexOptions,
284    ) -> Result<Self, anyhow::Error> {
285        let (meta_opt, index): (Option<FileMetadata>, Index) =
286            create_index(cache_dir.as_ref(), config, "filepath", vec!["contents"])?;
287        let meta = meta_opt.unwrap_or_else(|| FileMetadata::new(config.clone(), for_dir).unwrap());
288        let filepath = index
289            .schema()
290            .get_field("filepath")
291            .expect("No field named 'filepath'");
292        let contents = index
293            .schema()
294            .get_field("contents")
295            .expect("No field named 'contents'");
296        Ok(Self {
297            index,
298            cache_dir: cache_dir.map(|p| fs::canonicalize(p).unwrap()),
299            meta,
300            filepath,
301            contents,
302        })
303    }

305    /// Returns a configured directory walker for scanning the indexed directory.
306    ///
307    /// The walker respects the index's `.gitignore`, hidden file, glob, and
308    /// symlink-following settings.
309    pub fn get_file_walker(&self) -> Result<WalkBuilder, anyhow::Error> {
310        let mut builder = WalkBuilder::new(&self.meta.for_dir);
311        builder
312            .hidden(!self.meta.config.hidden)
313            .threads(self.meta.config.threads)
314            .ignore(self.meta.config.ignore_files)
315            .git_global(self.meta.config.ignore_files)
316            .git_ignore(self.meta.config.ignore_files)
317            .git_exclude(self.meta.config.ignore_files)
318            .follow_links(self.meta.config.follow);
319        if !self.meta.config.glob.is_empty() {
320            let mut globs = OverrideBuilder::new(&self.meta.for_dir);
321            globs.case_insensitive(self.meta.config.glob_case_insensitive)?;
322            for glob in &self.meta.config.glob {
323                globs.add(glob)?;
324            }
325            builder.overrides(globs.build()?);
326        }
327        if !self.meta.config.oglob.is_empty() {
328            let mut globs = OverrideBuilder::new(&self.meta.for_dir);
329            globs.case_insensitive(self.meta.config.glob_case_insensitive)?;
330            for glob in &self.meta.config.oglob {
331                globs.add(glob)?;
332            }
333            let matcher = globs.build()?;
334            builder.filter_entry(move |e| {
335                if e.file_type().map(|ft| ft.is_dir()).unwrap_or(false) {
336                    true
337                } else {
338                    matcher.matched(e.path(), false).is_whitelist()
339                }
340            });
341        }
342        Ok(builder)
343    }

345    /// Scans the indexed directory and adds/updates documents in the index.
346    ///
347    /// If `rebuild` is true, all readable files are re-indexed regardless of
348    /// their modification time. Otherwise, only files modified since the last
349    /// update are added.
350    pub fn update(&mut self, rebuild: bool) -> Result<&mut Self, anyhow::Error> {
351        let mut index_writer = self.index.writer::<tantivy::TantivyDocument>(50_000_000)?;
352        let walker = self.get_file_walker()?;
353        let now = Utc::now();
354        walker.build_parallel().run(|| {
355            Box::new(|result| {
356                if let Ok(entry) = result {
357                    if let Ok(contents) = fs::read_to_string(entry.path()) {
358                        let modified: DateTime<Utc> =
359                            entry.metadata().unwrap().modified().unwrap().into();
360                        if rebuild || modified > self.meta.last_update {
361                            let filepath = entry.path().strip_prefix(&self.meta.for_dir).unwrap();
362                            let doc = doc!(
363                                self.filepath => String::from(filepath.to_string_lossy()),
364                                self.contents => contents,
365                            );
366                            let _ = index_writer.add_document(doc);
367                        }
368                    }
369                }
370                WalkState::Continue
371            })
372        });
373
374        index_writer.commit()?;
375        self.meta.last_update = now;
376        if let Some(index_dir) = &self.cache_dir {
377            fs::write(
378                index_dir.join(METADATA_FILE),
379                serde_json::to_string(&self.meta)?,
380            )?;
381        }
382
383        Ok(self)
384    }

386    /// Executes a search query against the file index.
387    ///
388    /// Returns results sorted by relevance, limited by [`FileSearchOptions::limit`],
389    /// with matching line numbers and text when `filename_only` is false.
390    pub fn search(
391        &self,
392        query: &dyn Query,
393        opts: &FileSearchOptions,
394    ) -> Result<Vec<FileSearchResult>, anyhow::Error> {
395        let reader = self
396            .index
397            .reader_builder()
398            .reload_policy(ReloadPolicy::OnCommitWithDelay)
399            .try_into()?;
400        let searcher = reader.searcher();
401        let top_docs = searcher.search(query, &TopDocs::with_limit(opts.limit).order_by_score())?;
402        let mut doc_results = Vec::new();
403        for (score, doc_address) in top_docs {
404            if score > opts.threshold {
405                doc_results.push(DocResult {
406                    score,
407                    address: doc_address,
408                });
409            }
410        }
411        let mut position_map = location::get_search_results(self, query, &searcher, &doc_results)?;
412        let mut results = Vec::new();
413        for doc_result in doc_results {
414            let doc: tantivy::TantivyDocument = searcher.doc(doc_result.address)?;
415            let filepath = doc.get_first(*self.filepath()).unwrap().as_str().unwrap();
416            let fullpath = if let Some(root_dir) = opts.root_dir.as_deref() {
417                PathBuf::from(root_dir).join(filepath)
418            } else {
419                PathBuf::from(self.meta.for_dir()).join(filepath)
420            };
421
422            let mut lines = Vec::new();
423            if !opts.filename_only {
424                if let Some(position_data) = position_map.get_mut(&doc_result.address) {
425                    location::positions_to_lines(self, &fullpath, position_data, &mut lines)?
426                };
427            }
428            results.push(FileSearchResult {
429                file: fullpath,
430                score: doc_result.score,
431                lines,
432            });
433        }
434        Ok(results)
435    }

438impl Display for FileIndex {
439    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
440        writeln!(f, "Index({:?})", self.meta.for_dir)?;
441        writeln!(f, "  version: {}", self.meta.version)?;
442        if let Some(index_dir) = &self.cache_dir {
443            writeln!(f, "  location: {:?}", index_dir)?;
444            writeln!(
445                f,
446                "  last updated: {}",
447                DateTime::<Local>::from(self.meta.last_update)
448            )?;
449        } else {
450            writeln!(f, "  location: in-memory")?;
451        }
452        for field in serde_json::to_string_pretty(&self.meta.config)
453            .unwrap_or("".to_string())
454            .split("\n")
455        {
456            writeln!(f, "  {}", field.replace(" =", ":"))?;
457        }
458        Ok(())
459    }
