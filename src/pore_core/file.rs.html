1//! File-based full-text indexing.
2//!
3//! This module provides [`FileIndex`], which indexes the text contents of files
4//! in a directory tree. It uses the `ignore` crate for efficient parallel
5//! directory walking that respects `.gitignore`, `.gitexclude`, and other
6//! ignore files.
7//!
8//! # Key types
9//! - [`FileIndex`] — the index handle, created via [`FileIndex::get_or_create`].
10//! - [`FileIndexOptions`] — configuration for file walking and tokenization.
11//! - [`FileSearchOptions`] — parameters controlling search result limits and format.
12//! - [`FileMetadata`] — persisted metadata about when and how the index was built.
13//! - [`FileSearchResult`] and [`Line`] — search result structures.
14

88/// Configuration options for building a [`FileIndex`].
89///
90/// Controls which files are included, whether to follow symlinks, the stemming
91/// language, and how many threads to use for the directory walker.
92///
93/// The `#[create_option_copy]` macro generates a companion `*Shape` struct and
94/// copy-conversion functions for Lua interop.
95#[create_option_copy(FileIndexOptionsShape)]
96#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
97pub struct FileIndexOptions {
98    /// Whether to follow symbolic links during file walking.
99    pub follow: bool,
100    /// Glob patterns to include files.
101    pub glob: Vec<String>,
102    /// Whether glob matching is case-insensitive.
103    pub glob_case_insensitive: bool,
104    /// Whether to include hidden files and directories.
105    pub hidden: bool,
106    /// Whether to respect .gitignore and similar ignore files.
107    pub ignore_files: bool,
108    /// Language used for stemming.
109    pub language: LanguageRef,
110    /// Glob patterns to exclude files (takes precedence over `glob`).
111    pub oglob: Vec<String>,
112    // TODO: move this field to a more appropriate location.
113    /// Number of threads for parallel directory walking (0 = auto).
114    pub threads: usize,
115}
116

119        FileIndexOptions {

122            language: LanguageRef::English,
123            ignore_files: true,
124            glob_case_insensitive: false,
125            glob: vec![],

132/// Options that control search result formatting and limits.

228///
229/// Contains the file path, relevance score, and the match detail. By default the

340    /// Opens an existing index or creates a new one.
341    ///
342    /// If a persisted index exists at `cache_dir` with a matching config, it is
343    /// reused. Otherwise a fresh index is created.
344    ///
345    /// # Parameters
346    /// * `for_dir` — the directory to index.
347    /// * `cache_dir` — optional path for persisted index files.
348    /// * `config` — indexing options.
349    pub fn get_or_create<P: AsRef<Path>>(
350        for_dir: P,
351        cache_dir: Option<P>,
352        config: &FileIndexOptions,
353    ) -> Result<Self, anyhow::Error> {
354        let (meta_opt, index): (Option<FileMetadata>, Index) = create_index(
355            cache_dir.as_ref(),
356            config,
357            "filepath",
358            vec!["contents".to_string()],
359            true,
360        )?;
361        let meta = meta_opt.unwrap_or_else(|| FileMetadata::new(config.clone(), for_dir).unwrap());
362        let filepath = index
363            .schema()
364            .get_field("filepath")
365            .expect("No field named 'filepath'");
366        let contents = index
367            .schema()
368            .get_field("contents")
369            .expect("No field named 'contents'");
370        let modified = index
371            .schema()
372            .get_field("modified")
373            .expect("No field named 'modified'");
374        let ext = index
375            .schema()
376            .get_field("ext")
377            .expect("No field named 'ext'");
378        Ok(Self {
379            index,
380            cache_dir: cache_dir.map(|p| fs::canonicalize(p).unwrap()),
381            meta,
382            filepath,
383            contents,
384            modified,
385            ext,
386        })
387    }

389    /// Returns a configured directory walker for scanning the indexed directory.
390    ///
391    /// The walker respects the index's `.gitignore`, hidden file, glob, and
392    /// symlink-following settings.
393    pub fn get_file_walker(&self) -> Result<WalkBuilder, anyhow::Error> {
394        let mut builder = WalkBuilder::new(&self.meta.for_dir);
395        builder
396            .hidden(!self.meta.config.hidden)
397            .threads(self.meta.config.threads)
398            .ignore(self.meta.config.ignore_files)
399            .git_global(self.meta.config.ignore_files)
400            .git_ignore(self.meta.config.ignore_files)
401            .git_exclude(self.meta.config.ignore_files)
402            .follow_links(self.meta.config.follow);
403        if !self.meta.config.glob.is_empty() {
404            let mut globs = OverrideBuilder::new(&self.meta.for_dir);
405            globs.case_insensitive(self.meta.config.glob_case_insensitive)?;
406            for glob in &self.meta.config.glob {
407                globs.add(glob)?;
408            }
409            builder.overrides(globs.build()?);
410        }
411        if !self.meta.config.oglob.is_empty() {
412            let mut globs = OverrideBuilder::new(&self.meta.for_dir);
413            globs.case_insensitive(self.meta.config.glob_case_insensitive)?;
414            for glob in &self.meta.config.oglob {
415                globs.add(glob)?;
416            }
417            let matcher = globs.build()?;
418            builder.filter_entry(move |e| {
419                if e.file_type().map(|ft| ft.is_dir()).unwrap_or(false) {
420                    true
421                } else {
422                    matcher.matched(e.path(), false).is_whitelist()
423                }
424            });
425        }
426        Ok(builder)
427    }

429    /// Scans the indexed directory and adds/updates documents in the index.
430    ///
431    /// If `rebuild` is true, all readable files are re-indexed regardless of
432    /// their modification time. Otherwise, only files modified since the last
433    /// update are added.
434    pub fn update(&mut self, rebuild: bool) -> Result<&mut Self, anyhow::Error> {
435        let mut index_writer = self.index.writer::<PoreFileEntry>(50_000_000)?;
436        let walker = self.get_file_walker()?;
437        let now = Utc::now();
438        walker.build_parallel().run(|| {
439            Box::new(|result| {
440                if let Ok(entry) = result {
441                    if let Ok(contents) = fs::read_to_string(entry.path()) {
442                        let modified: DateTime<Utc> =
443                            entry.metadata().unwrap().modified().unwrap().into();
444                        if rebuild || modified > self.meta.last_update {
445                            let filepath = entry.path().strip_prefix(&self.meta.for_dir).unwrap();
446                            let filepath_str = filepath.to_string_lossy().into_owned();
447                            let ext_str = filepath
448                                .extension()
449                                .unwrap_or_default()
450                                .to_string_lossy()
451                                .into_owned();
452                            let doc = PoreFileEntry {
453                                filepath_field: self.filepath,
454                                filepath_value: OwnedValue::Str(filepath_str),
455                                contents_field: self.contents,
456                                contents_value: OwnedValue::Str(contents),
457                                modified_field: self.modified,
458                                modified_value: OwnedValue::U64(modified.timestamp_millis() as u64),
459                                ext_field: self.ext,
460                                ext_value: OwnedValue::Str(ext_str),
461                            };
462                            let _ = index_writer.add_document(doc);
463                        }
464                    }
465                }
466                WalkState::Continue
467            })
468        });
469
470        index_writer.commit()?;
471        self.meta.last_update = now;
472        if let Some(index_dir) = &self.cache_dir {
473            fs::write(
474                index_dir.join(METADATA_FILE),
475                serde_json::to_string(&self.meta)?,
476            )?;
477        }
478
479        Ok(self)
480    }

482    /// Groups the documents matching `query` into buckets by a fast field.
483    ///
484    /// The field named by [`FileSearchOptions::aggregate`] must exist in the schema and be
485    /// a FAST field — `ext`, `filepath` and `modified` qualify. Returns Tantivy's terms
486    /// aggregation result as JSON, shaped `{"<field>": {"buckets": [{"key", "doc_count"}]}}`.
487    ///
488    /// # Errors
489    ///
490    /// Returns an error naming the field when it is absent from the schema, is not a fast
491    /// field, or when no aggregate field was requested at all.
492    pub fn aggregate(
493        &self,
494        query: &dyn Query,
495        opts: &FileSearchOptions,
496    ) -> Result<serde_json::Value, anyhow::Error> {
497        use tantivy::aggregation::agg_req::{Aggregation, AggregationVariants, Aggregations};
498        use tantivy::aggregation::bucket::TermsAggregation;
499        use tantivy::aggregation::{AggContextParams, AggregationCollector};
500
501        let field_name = opts
502            .aggregate
503            .as_deref()
504            .ok_or_else(|| anyhow::anyhow!("no aggregate field requested"))?;
505
506        // Fail with the field name rather than letting Tantivy raise a generic error deep
507        // inside the collector.
508        let schema = self.index.schema();
509        let field = schema
510            .get_field(field_name)
511            .map_err(|_| anyhow::anyhow!("cannot aggregate on unknown field '{field_name}'"))?;
512        if !schema.get_field_entry(field).is_fast() {
513            return Err(anyhow::anyhow!(
514                "cannot aggregate on '{field_name}': it is not a fast field"
515            ));
516        }
517
518        let mut aggs = Aggregations::default();
519        aggs.insert(
520            field_name.to_string(),
521            Aggregation {
522                agg: AggregationVariants::Terms(TermsAggregation {
523                    field: field_name.to_string(),
524                    ..Default::default()
525                }),
526                sub_aggregation: Aggregations::default(),
527            },
528        );
529
530        let searcher = self.index.reader()?.searcher();
531        let collector = AggregationCollector::from_aggs(
532            aggs,
533            AggContextParams::new(Default::default(), self.index.tokenizers().clone()),
534        );
535        Ok(serde_json::to_value(searcher.search(query, &collector)?)?)
536    }

538    /// Executes a search query against the file index.
539    ///
540    /// Returns results sorted by relevance, limited by [`FileSearchOptions::limit`],
541    /// with matching line numbers and text when `filename_only` is false.
542    pub fn search(
543        &self,
544        query: &dyn Query,
545        opts: &FileSearchOptions,
546    ) -> Result<Vec<FileSearchResult>, anyhow::Error> {
547        let reader = self
548            .index
549            .reader_builder()
550            .reload_policy(ReloadPolicy::OnCommitWithDelay)
551            .try_into()?;
552        let searcher = reader.searcher();
553
554        let process_docs = |doc_addresses: Vec<tantivy::DocAddress>,
555                            scores: Vec<f32>|
556         -> Result<Vec<FileSearchResult>, anyhow::Error> {
557            let mut doc_results = Vec::new();
558            for (i, address) in doc_addresses.into_iter().enumerate() {
559                let score = scores.get(i).copied().unwrap_or(0.0);
560                // Sorted collectors report a score of 0.0, so the threshold only
561                // applies to relevance-ordered results.
562                if opts.sort.is_none() && score <= opts.threshold {
563                    continue;
564                }
565                doc_results.push(DocResult { score, address });
566            }
567
568            let want_detail = !opts.filename_only;
569            let snippet_generator = if want_detail && opts.snippets {
570                Some(tantivy::snippet::SnippetGenerator::create(
571                    &searcher,
572                    query,
573                    *self.contents(),
574                )?)
575            } else {
576                None
577            };
578            let mut position_map = if want_detail && !opts.snippets {
579                Some(location::get_search_results(
580                    self,
581                    query,
582                    &searcher,
583                    &doc_results,
584                )?)
585            } else {
586                None
587            };
588
589            let mut res = Vec::new();
590            for doc_result in doc_results {
591                let doc: tantivy::TantivyDocument = searcher.doc(doc_result.address)?;
592                let filepath = doc.get_first(*self.filepath()).unwrap().as_str().unwrap();
593                let fullpath = if let Some(root_dir) = opts.root_dir.as_deref() {
594                    PathBuf::from(root_dir).join(filepath)
595                } else {
596                    PathBuf::from(self.meta.for_dir()).join(filepath)
597                };
598
599                if let Some(generator) = &snippet_generator {
600                    let snippet = generator.snippet_from_doc(&doc);
601                    let mut snippets = Vec::new();
602                    if !snippet.fragment().is_empty() {
603                        snippets.push(snippet.to_html());
604                    }
605                    res.push(FileSearchResult::with_snippets(
606                        fullpath,
607                        doc_result.score,
608                        snippets,
609                    ));
610                } else {
611                    let mut lines = Vec::new();
612                    if let Some(position_data) = position_map
613                        .as_mut()
614                        .and_then(|map| map.get_mut(&doc_result.address))
615                    {
616                        location::positions_to_lines(self, &fullpath, position_data, &mut lines)?;
617                    }
618                    res.push(FileSearchResult::new(fullpath, doc_result.score, lines));
619                }
620            }
621            Ok(res)
622        };
623
624        if let Some(sort_field) = &opts.sort {
625            if sort_field == "date" {
626                let top_docs = searcher.search(
627                    query,
628                    &tantivy::collector::TopDocs::with_limit(opts.limit)
629                        .order_by_fast_field::<u64>("modified", tantivy::Order::Desc),
630                )?;
631                let len = top_docs.len();
632                process_docs(top_docs.into_iter().map(|x| x.1).collect(), vec![0.0; len])
633            } else if sort_field == "path" {
634                let top_docs = searcher.search(
635                    query,
636                    &tantivy::collector::TopDocs::with_limit(opts.limit)
637                        .order_by_string_fast_field("filepath", tantivy::Order::Asc),
638                )?;
639                let len = top_docs.len();
640                process_docs(top_docs.into_iter().map(|x| x.1).collect(), vec![0.0; len])
641            } else {
642                let top_docs = searcher.search(
643                    query,
644                    &tantivy::collector::TopDocs::with_limit(opts.limit).order_by_score(),
645                )?;
646                process_docs(
647                    top_docs.iter().map(|x| x.1).collect(),
648                    top_docs.iter().map(|x| x.0).collect(),
649                )
650            }
651        } else {
652            let top_docs = searcher.search(
653                query,
654                &tantivy::collector::TopDocs::with_limit(opts.limit).order_by_score(),
655            )?;
656            process_docs(
657                top_docs.iter().map(|x| x.1).collect(),
658                top_docs.iter().map(|x| x.0).collect(),
659            )
660        }
661    }
662}

664impl Display for FileIndex {
665    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
666        writeln!(f, "Index({:?})", self.meta.for_dir)?;
667        writeln!(f, "  version: {}", self.meta.version)?;
668        if let Some(index_dir) = &self.cache_dir {
669            writeln!(f, "  location: {:?}", index_dir)?;
670            writeln!(
671                f,
672                "  last updated: {}",
673                DateTime::<Local>::from(self.meta.last_update)
674            )?;
675        } else {
676            writeln!(f, "  location: in-memory")?;
677        }
678        for field in serde_json::to_string_pretty(&self.meta.config)
679            .unwrap_or("".to_string())
680            .split("\n")
681        {
682            writeln!(f, "  {}", field.replace(" =", ":"))?;
683        }
684        Ok(())
685    }
686}
