1//! jq filter compilation and evaluation engine.
2//!
3//! Wraps [`jaq_core`] to provide a simple compile-then-run API for jq expressions.
4
5use jaq_core::load::{Arena, File, Loader};
6use jaq_core::{data, unwrap_valr, Compiler, Ctx, Vars};
7use jaq_json::Val;
8
9/// A compiled jq filter that can be run against JSON values.
10pub struct JqEngine {
11    filter: jaq_core::Filter<data::JustLut<Val>>,
12}
13
14impl JqEngine {
15    /// Compile a jq filter string.
16    pub fn compile(filter_str: &str) -> Result<Self, anyhow::Error> {
17        let program = File {
18            code: filter_str,
19            path: (),
20        };
21        let defs = jaq_core::defs().chain(jaq_std::defs());
22        let funs = jaq_core::funs::<data::JustLut<Val>>().chain(jaq_std::funs());
23
24        let loader = Loader::new(defs);
25        let arena = Arena::default();
26        let modules = loader
27            .load(&arena, program)
28            .map_err(|errs| anyhow::anyhow!("jq load error: {:?}", errs))?;
29
30        let filter = Compiler::default()
31            .with_funs(funs)
32            .compile(modules)
33            .map_err(|errs| anyhow::anyhow!("jq compile error: {:?}", errs))?;
34
35        Ok(Self { filter })
36    }
37
38    /// Run the compiled filter against a JSON value.
39    pub fn run(&self, input: &serde_json::Value) -> Result<Vec<serde_json::Value>, anyhow::Error> {
40        // Convert serde_json::Value to jaq_json::Val using serde feature
41        let input_val: Val = serde_json::from_value(input.clone())
42            .map_err(|e| anyhow::anyhow!("failed to convert json to jaq: {}", e))?;
43
44        let ctx = Ctx::<data::JustLut<Val>>::new(&self.filter.lut, Vars::new([]));
45        let out = self.filter.id.run((ctx, input_val)).map(unwrap_valr);
46
47        let mut results = Vec::new();
48        for output in out {
49            match output {
50                Ok(val) => {
51                    let json_val: serde_json::Value = serde_json::from_str(&val.to_string())
52                        .map_err(|e| {
53                            anyhow::anyhow!("failed to parse jaq output as json: {}", e)
54                        })?;
55                    results.push(json_val);
56                }
57                Err(e) => {
58                    return Err(anyhow::anyhow!("jq runtime error: {:?}", e));
59                }
60            }
61        }
62        Ok(results)
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use serde_json::json;
70
71    #[test]
72    fn compile_valid_filter() {
73        let engine = JqEngine::compile(".foo");
74        assert!(engine.is_ok());
75    }
76
77    #[test]
78    fn compile_invalid_filter() {
79        let engine = JqEngine::compile(".[invalid syntax!!");
80        assert!(engine.is_err());
81    }
82
83    #[test]
84    fn run_identity_filter() {
85        let engine = JqEngine::compile(".").unwrap();
86        let input = json!({"a": 1, "b": 2});
87        let results = engine.run(&input).unwrap();
88        assert_eq!(results.len(), 1);
89        assert_eq!(results[0], json!({"a": 1, "b": 2}));
90    }
91
92    #[test]
93    fn run_field_access() {
94        let engine = JqEngine::compile(".name").unwrap();
95        let input = json!({"name": "pore", "version": "0.2.0"});
96        let results = engine.run(&input).unwrap();
97        assert_eq!(results, vec![json!("pore")]);
98    }
99
100    #[test]
101    fn run_array_filter() {
102        let engine = JqEngine::compile("[.[] | select(. > 2)]").unwrap();
103        let input = json!([1, 2, 3, 4, 5]);
104        let results = engine.run(&input).unwrap();
105        assert_eq!(results, vec![json!([3, 4, 5])]);
106    }
107
108    #[test]
109    fn run_multiple_outputs() {
110        let engine = JqEngine::compile(".[]").unwrap();
111        let input = json!([10, 20, 30]);
112        let results = engine.run(&input).unwrap();
113        assert_eq!(results, vec![json!(10), json!(20), json!(30)]);
114    }
115
116    #[test]
117    fn run_empty_output() {
118        let engine = JqEngine::compile("empty").unwrap();
119        let input = json!(null);
120        let results = engine.run(&input).unwrap();
121        assert!(results.is_empty());
122    }
123
124    #[test]
125    fn run_string_interpolation() {
126        let engine = JqEngine::compile(r#""\(.a):\(.b)""#).unwrap();
127        let input = json!({"a": "hello", "b": "world"});
128        let results = engine.run(&input).unwrap();
129        assert_eq!(results, vec![json!("hello:world")]);
130    }
131
132    #[test]
133    fn run_sort_by() {
134        let engine = JqEngine::compile("sort_by(.x)").unwrap();
135        let input = json!([{"x": 3}, {"x": 1}, {"x": 2}]);
136        let results = engine.run(&input).unwrap();
137        assert_eq!(results, vec![json!([{"x": 1}, {"x": 2}, {"x": 3}])]);
138    }
139}
